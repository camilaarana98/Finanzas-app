// ============================================================ // FINANZAS PERSONALES — Google Apps Script v3 // Pega este código en Extensiones > Apps Script de tu Sheet // ============================================================ const SHEET_CONFIG = "Configuración"; const SHEET_HISTORIAL = "Historial"; const SHEET_CFG_MES = "Config Mensual"; // creada automáticamente al primer uso function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ─── ROUTER ────────────────────────────────────────────────── // Todo va por doGet para evitar problemas de CORS. function doGet(e) { var action = ((e.parameter && e.parameter.action) || "get").toLowerCase(); try { switch (action) { case "get": return handleGet(e); case "post": return handlePost(e); case "delete": return handleDelete(e); case "history": return handleHistory(e); case "months": return handleMonths(e); case "togglefijo": return handleToggleFijo(e); case "updatemonthconfig": return handleUpdateMonthConfig(e); case "updateconfig": return handleUpdateConfig(e); case "addfijo": return handleAddFijo(e); case "editfijo": return handleEditFijo(e); case "deletefijo": return handleDeleteFijo(e); default: return jsonResponse({ ok: false, error: "Acción no reconocida: " + action }); } } catch (err) { return jsonResponse({ ok: false, error: err.message }); } } // ─── HELPERS ───────────────────────────────────────────────── // Devuelve { ingreso, ahorro } para el mes dado. // Primero busca en "Config Mensual"; si no hay, usa la Config global. function getMonthConfig(ss, mes, anio) { var key = anio + "-" + (mes < 10 ? "0" + mes : "" + mes); var cfgMSheet = ss.getSheetByName(SHEET_CFG_MES); if (cfgMSheet) { var data = cfgMSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === key) { return { ingreso: Number(data[i][1]) || 0, ahorro: Number(data[i][2]) || 0 }; } } } var cfgSheet = ss.getSheetByName(SHEET_CONFIG); var cfgData = cfgSheet ? cfgSheet.getDataRange().getValues() : []; return { ingreso: cfgData.length > 1 ? Number(cfgData[1][1]) || 0 : 0, ahorro: cfgData.length > 2 ? Number(cfgData[2][1]) || 0 : 0 }; } // Lee los gastos fijos de la pestaña Configuración. function getGastosFijos(ss) { var cfgSheet = ss.getSheetByName(SHEET_CONFIG); if (!cfgSheet) return []; var cfgData = cfgSheet.getDataRange().getValues(); var result = []; for (var i = 5; i < cfgData.length; i++) { var nombre = cfgData[i][0]; if (!nombre) break; result.push({ nombre: nombre, monto: Number(cfgData[i][1]) || 0, tipo: cfgData[i][2] || "Manual" }); } return result; } // Devuelve un objeto { "Nombre gasto": true } para los fijos marcados como pagados // en el mes/año indicado. function getFijosPagados(histData, mes, anio) { var pagados = {}; for (var i = 1; i < histData.length; i++) { var fila = histData[i]; if (!fila[0] || fila[3] !== "FijoPagado") continue; var fecha = new Date(fila[0]); if (fecha.getMonth() + 1 === mes && fecha.getFullYear() === anio) { pagados[fila[2]] = true; } } return pagados; } // ─── GET: dashboard del mes actual ─────────────────────────── function handleGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var now = new Date(); var mes = now.getMonth() + 1; var anio = now.getFullYear(); var cfg = getMonthConfig(ss, mes, anio); var gastosFijos = getGastosFijos(ss); var totalFijos = gastosFijos.reduce(function(s, g) { return s + g.monto; }, 0); var saldoDisp = cfg.ingreso - cfg.ahorro - totalFijos; var histSheet = ss.getSheetByName(SHEET_HISTORIAL); if (!histSheet) throw new Error("Pestaña '" + SHEET_HISTORIAL + "' no encontrada."); var histData = histSheet.getDataRange().getValues(); var fijosPagados = getFijosPagados(histData, mes, anio); gastosFijos = gastosFijos.map(function(g) { return { nombre: g.nombre, monto: g.monto, tipo: g.tipo, pagado: !!fijosPagados[g.nombre] }; }); var totalVar = 0, totalAho = 0; var historial = []; for (var i = 1; i < histData.length; i++) { var fila = histData[i]; if (!fila[0]) continue; var tipo = fila[3] || "Variable"; if (tipo === "FijoPagado") continue; // marcadores internos, ignorar var fecha = new Date(fila[0]); if (fecha.getMonth() + 1 !== mes || fecha.getFullYear() !== anio) continue; var monto = Number(fila[1]) || 0; historial.push({ rowIndex: i + 1, fecha: fila[0], monto: monto, descripcion: fila[2] || "", tipoGasto: tipo, categoria: fila[4] || "" }); if (tipo === "Ahorros") totalAho += monto; else totalVar += monto; } historial.sort(function(a, b) { return new Date(b.fecha) - new Date(a.fecha); }); return jsonResponse({ ok: true, ingreso: cfg.ingreso, ahorro: cfg.ahorro, gastosFijos: gastosFijos, totalFijos: totalFijos, saldoDisponible: saldoDisp, totalVariables: totalVar, totalDeAhorros: totalAho, saldoRestante: saldoDisp - totalVar, historial: historial }); } // ─── POST: registrar nuevo gasto ───────────────────────────── function handlePost(e) { var monto = Number(e.parameter.monto); var descripcion = e.parameter.descripcion || ""; var tipoGasto = e.parameter.tipoGasto || "Variable"; var categoria = e.parameter.categoria || "Otros"; var fechaParam = e.parameter.fecha || ""; // "YYYY-MM-DD" opcional if (!monto || monto <= 0) throw new Error("Monto inválido."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var histSheet = ss.getSheetByName(SHEET_HISTORIAL); if (!histSheet) throw new Error("Pestaña '" + SHEET_HISTORIAL + "' no encontrada."); var fechaStr; if (fechaParam && fechaParam.match(/^\d{4}-\d{2}-\d{2}$/)) { var parts = fechaParam.split("-"); var d = new Date( Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]), new Date().getHours(), new Date().getMinutes(), 0 ); fechaStr = Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); } else { fechaStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); } histSheet.appendRow([fechaStr, monto, descripcion, tipoGasto, categoria]); return jsonResponse({ ok: true, mensaje: "Gasto registrado." }); } // ─── DELETE: eliminar fila por número de fila ───────────────── function handleDelete(e) { var rowIndex = Number(e.parameter.rowIndex); if (!rowIndex || rowIndex < 2) throw new Error("Índice de fila inválido."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var histSheet = ss.getSheetByName(SHEET_HISTORIAL); if (!histSheet) throw new Error("Pestaña '" + SHEET_HISTORIAL + "' no encontrada."); if (rowIndex > histSheet.getLastRow()) throw new Error("Fila no encontrada."); histSheet.deleteRow(rowIndex); return jsonResponse({ ok: true }); } // ─── HISTORY: datos de un mes específico ───────────────────── function handleHistory(e) { var mes = Number(e.parameter.mes); var anio = Number(e.parameter.anio); if (!mes || !anio) throw new Error("Parámetros 'mes' y 'anio' requeridos."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var cfg = getMonthConfig(ss, mes, anio); var gastosFijos = getGastosFijos(ss); var totalFijos = gastosFijos.reduce(function(s, g) { return s + g.monto; }, 0); var saldoDisp = cfg.ingreso - cfg.ahorro - totalFijos; var histSheet = ss.getSheetByName(SHEET_HISTORIAL); if (!histSheet) throw new Error("Pestaña '" + SHEET_HISTORIAL + "' no encontrada."); var histData = histSheet.getDataRange().getValues(); var fijosPagados = getFijosPagados(histData, mes, anio); gastosFijos = gastosFijos.map(function(g) { return { nombre: g.nombre, monto: g.monto, tipo: g.tipo, pagado: !!fijosPagados[g.nombre] }; }); var totalVar = 0, totalAho = 0; var historial = []; for (var i = 1; i < histData.length; i++) { var fila = histData[i]; if (!fila[0]) continue; var tipo = fila[3] || "Variable"; if (tipo === "FijoPagado") continue; var fecha = new Date(fila[0]); if (fecha.getMonth() + 1 !== mes || fecha.getFullYear() !== anio) continue; var monto = Number(fila[1]) || 0; historial.push({ rowIndex: i + 1, fecha: fila[0], monto: monto, descripcion: fila[2] || "", tipoGasto: tipo, categoria: fila[4] || "" }); if (tipo === "Ahorros") totalAho += monto; else totalVar += monto; } historial.sort(function(a, b) { return new Date(b.fecha) - new Date(a.fecha); }); return jsonResponse({ ok: true, ingreso: cfg.ingreso, ahorro: cfg.ahorro, gastosFijos: gastosFijos, totalFijos: totalFijos, saldoDisponible: saldoDisp, totalVariables: totalVar, totalDeAhorros: totalAho, saldoRestante: saldoDisp - totalVar, historial: historial }); } // ─── MONTHS: lista de meses disponibles ────────────────────── // Siempre incluye el mes actual + 2 meses siguientes (para registros anticipados). function handleMonths(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var histSheet = ss.getSheetByName(SHEET_HISTORIAL); var monthSet = {}; // Mes actual y los dos siguientes var now = new Date(); for (var offset = 0; offset <= 2; offset++) { var d = new Date(now.getFullYear(), now.getMonth() + offset, 1); var m = d.getMonth() + 1; var a = d.getFullYear(); var key = a + "-" + (m < 10 ? "0" + m : "" + m); monthSet[key] = { mes: m, anio: a }; } // Meses con datos en Historial if (histSheet) { var histData = histSheet.getDataRange().getValues(); for (var i = 1; i < histData.length; i++) { var fila = histData[i]; if (!fila[0]) continue; var fecha = new Date(fila[0]); var m2 = fecha.getMonth() + 1; var a2 = fecha.getFullYear(); var key2 = a2 + "-" + (m2 < 10 ? "0" + m2 : "" + m2); monthSet[key2] = { mes: m2, anio: a2 }; } } var months = Object.keys(monthSet) .sort(function(a, b) { return b.localeCompare(a); }) .map(function(k) { return monthSet[k]; }); return jsonResponse({ ok: true, months: months }); } // ─── TOGGLE FIJO: marcar/desmarcar un gasto fijo como pagado ─ function handleToggleFijo(e) { var nombre = e.parameter.nombre; var mes = Number(e.parameter.mes); var anio = Number(e.parameter.anio); var paid = e.parameter.paid === "true"; if (!nombre || !mes || !anio) throw new Error("Parámetros incompletos."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var histSheet = ss.getSheetByName(SHEET_HISTORIAL); if (!histSheet) throw new Error("Pestaña '" + SHEET_HISTORIAL + "' no encontrada."); var histData = histSheet.getDataRange().getValues(); if (paid) { // Verificar si ya está marcado for (var i = 1; i < histData.length; i++) { var fila = histData[i]; if (fila[3] !== "FijoPagado" || fila[2] !== nombre) continue; var f = new Date(fila[0]); if (f.getMonth() + 1 === mes && f.getFullYear() === anio) { return jsonResponse({ ok: true }); // ya estaba marcado } } // Agregar fila marcadora (día 15 del mes para ordenamiento neutral) var d = new Date(anio, mes - 1, 15); var fechaStr = Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); histSheet.appendRow([fechaStr, 0, nombre, "FijoPagado", "Fijo"]); } else { // Buscar y eliminar (de abajo hacia arriba para que el índice no cambie) for (var i = histData.length - 1; i >= 1; i--) { var fila = histData[i]; if (fila[3] !== "FijoPagado" || fila[2] !== nombre) continue; var f = new Date(fila[0]); if (f.getMonth() + 1 === mes && f.getFullYear() === anio) { histSheet.deleteRow(i + 1); break; } } } return jsonResponse({ ok: true }); } // ─── UPDATE MONTH CONFIG: editar ingreso/ahorro de un mes ──── function handleUpdateMonthConfig(e) { var mes = Number(e.parameter.mes); var anio = Number(e.parameter.anio); var campo = e.parameter.campo; // "ingreso" | "ahorro" var valor = Number(e.parameter.valor); if (!mes || !anio || !campo || isNaN(valor) || valor < 0) throw new Error("Parámetros inválidos."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var key = anio + "-" + (mes < 10 ? "0" + mes : "" + mes); // Crear hoja si no existe var cfgMSheet = ss.getSheetByName(SHEET_CFG_MES); if (!cfgMSheet) { cfgMSheet = ss.insertSheet(SHEET_CFG_MES); cfgMSheet.getRange(1, 1, 1, 3).setValues([["Mes", "Ingreso", "Ahorro"]]); cfgMSheet.getRange("A1:C1").setFontWeight("bold").setBackground("#e8f0fe"); cfgMSheet.setColumnWidth(1, 100); cfgMSheet.setColumnWidth(2, 120); cfgMSheet.setColumnWidth(3, 120); } var data = cfgMSheet.getDataRange().getValues(); // Actualizar fila existente for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === key) { if (campo === "ingreso") cfgMSheet.getRange(i + 1, 2).setValue(valor); if (campo === "ahorro") cfgMSheet.getRange(i + 1, 3).setValue(valor); return jsonResponse({ ok: true }); } } // Crear nueva fila con defaults del mes y sobreescribir el campo var globalCfg = getMonthConfig(ss, mes, anio); var newRow = [key, globalCfg.ingreso, globalCfg.ahorro]; if (campo === "ingreso") newRow[1] = valor; if (campo === "ahorro") newRow[2] = valor; cfgMSheet.appendRow(newRow); return jsonResponse({ ok: true }); } // ─── UPDATE GLOBAL CONFIG: ingreso / ahorro por defecto ────── function handleUpdateConfig(e) { var campo = e.parameter.campo; var valor = Number(e.parameter.valor); if (!campo || isNaN(valor) || valor < 0) throw new Error("Parámetros inválidos."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var cfgSheet = ss.getSheetByName(SHEET_CONFIG); if (!cfgSheet) throw new Error("Hoja de Configuración no encontrada."); if (campo === "ingreso") cfgSheet.getRange(2, 2).setValue(valor); if (campo === "ahorro") cfgSheet.getRange(3, 2).setValue(valor); return jsonResponse({ ok: true }); } // ─── ADD FIJO ───────────────────────────────────────────────── function handleAddFijo(e) { var nombre = (e.parameter.nombre || "").trim(); var monto = Number(e.parameter.monto); var tipo = e.parameter.tipo || "Manual"; if (!nombre || !monto || monto <= 0) throw new Error("Parámetros inválidos."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var cfgSheet = ss.getSheetByName(SHEET_CONFIG); if (!cfgSheet) throw new Error("Hoja de Configuración no encontrada."); var lastRow = 5; while (cfgSheet.getRange(lastRow + 1, 1).getValue()) lastRow++; cfgSheet.getRange(lastRow + 1, 1).setValue(nombre); cfgSheet.getRange(lastRow + 1, 2).setValue(monto); cfgSheet.getRange(lastRow + 1, 3).setValue(tipo); return jsonResponse({ ok: true }); } // ─── EDIT FIJO ──────────────────────────────────────────────── function handleEditFijo(e) { var nombreOld = (e.parameter.nombreOld || "").trim(); var nombre = (e.parameter.nombre || "").trim(); var monto = Number(e.parameter.monto); var tipo = e.parameter.tipo || "Manual"; if (!nombreOld || !nombre || !monto || monto <= 0) throw new Error("Parámetros inválidos."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var cfgSheet = ss.getSheetByName(SHEET_CONFIG); if (!cfgSheet) throw new Error("Hoja de Configuración no encontrada."); var data = cfgSheet.getDataRange().getValues(); for (var i = 5; i < data.length; i++) { if (String(data[i][0]) === nombreOld) { cfgSheet.getRange(i + 1, 1).setValue(nombre); cfgSheet.getRange(i + 1, 2).setValue(monto); cfgSheet.getRange(i + 1, 3).setValue(tipo); return jsonResponse({ ok: true }); } } throw new Error("Gasto fijo '" + nombreOld + "' no encontrado."); } // ─── DELETE FIJO ────────────────────────────────────────────── function handleDeleteFijo(e) { var nombre = (e.parameter.nombre || "").trim(); if (!nombre) throw new Error("Nombre requerido."); var ss = SpreadsheetApp.getActiveSpreadsheet(); var cfgSheet = ss.getSheetByName(SHEET_CONFIG); if (!cfgSheet) throw new Error("Hoja de Configuración no encontrada."); var data = cfgSheet.getDataRange().getValues(); for (var i = data.length - 1; i >= 5; i--) { if (String(data[i][0]) === nombre) { cfgSheet.deleteRow(i + 1); return jsonResponse({ ok: true }); } } throw new Error("Gasto fijo '" + nombre + "' no encontrado."); } // ─── INIT (ejecutar UNA VEZ desde el editor) ────────────────── function inicializarSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var cfg = ss.getSheetByName(SHEET_CONFIG) || ss.insertSheet(SHEET_CONFIG); cfg.clearContents(); cfg.getRange(1, 1).setValue("CONFIGURACIÓN MENSUAL"); cfg.getRange(2, 1, 2, 2).setValues([ ["Ingreso mensual", 2500], ["Monto a ahorrar", 500] ]); cfg.getRange(5, 1, 1, 3).setValues([["GASTOS FIJOS", "Monto", "Tipo"]]); cfg.getRange(6, 1, 5, 3).setValues([ ["Alquiler", 800, "Manual"], ["Internet", 50, "Débito"], ["Netflix", 15, "Débito"], ["Seguro médico", 100, "Débito"], ["Gimnasio", 40, "Manual"] ]); cfg.getRange("A1").setFontWeight("bold").setFontSize(12); cfg.getRange("A5:C5").setFontWeight("bold").setBackground("#e8f0fe"); cfg.setColumnWidth(1, 200); cfg.setColumnWidth(2, 120); cfg.setColumnWidth(3, 120); var hist = ss.getSheetByName(SHEET_HISTORIAL) || ss.insertSheet(SHEET_HISTORIAL); hist.clearContents(); hist.getRange(1, 1, 1, 5).setValues([["Fecha", "Monto", "Descripción", "Tipo", "Categoría"]]); hist.getRange("A1:E1").setFontWeight("bold").setBackground("#e8f0fe"); hist.setColumnWidth(1, 160); hist.setColumnWidth(2, 100); hist.setColumnWidth(3, 200); hist.setColumnWidth(4, 100); hist.setColumnWidth(5, 140); SpreadsheetApp.getUi().alert("✅ Estructura creada.\n\nEdita los valores en 'Configuración' antes de usar la app.");
